        function renderAdhkarList() {
            const container = document.getElementById('adhkar-list-container');
            const list = adhkarData[state.activeAdhkarType];
            container.innerHTML = '';
            let totalTarget = 0, currentTotal = 0;

            list.forEach((item, index) => {
                if(state.adhkarCounts[index] === undefined) state.adhkarCounts[index] = 0;
                totalTarget += item.count;
                currentTotal += state.adhkarCounts[index];
                const isCompleted = state.adhkarCounts[index] >= item.count;
                const progressPercent = (state.adhkarCounts[index] / item.count) * 100;
                
                const div = document.createElement('div');
                div.className = `adhkar-list-item bg-white dark:bg-slate-700 rounded-xl shadow-sm border border-gray-200 dark:border-gray-600 p-4 relative ${isCompleted ? 'completed' : ''}`;
                div.onclick = () => handleAdhkarClick(index);
                div.id = `adhkar-item-${index}`;
                div.innerHTML = `
                    <div class="adhkar-progress-fill absolute bottom-0 right-0 top-0 bg-secondary/10 dark:bg-secondary/20 z-0" style="width:${isCompleted ? '100%' : progressPercent + '%'}"></div>
                    <div class="relative z-10 flex gap-4 items-start">
                        <div class="flex-grow">
                            <p class="text-xl md:text-2xl font-heading leading-loose text-primary dark:text-[#fbbf24] text-justify">${item.text}</p>
                        </div>
                        <div class="flex-shrink-0 flex flex-col items-center justify-center min-w-[50px]">
                            <div class="w-12 h-12 rounded-full border-4 ${isCompleted ? 'border-secondary bg-secondary/10 text-secondary' : 'border-gray-200 dark:border-gray-500 text-gray-400'} flex items-center justify-center font-bold text-lg transition-colors duration-300" id="adhkar-counter-circle-${index}">${state.adhkarCounts[index]}</div>
                            <span class="text-xs text-gray-400 mt-1">${item.count}</span>
                        </div>
                    </div>
                `;
                container.appendChild(div);
            });
            updateTotalProgress(currentTotal, totalTarget);
        }

        function handleAdhkarClick(index) {
            const list = adhkarData[state.activeAdhkarType];
            const item = list[index];
            if (state.adhkarCounts[index] < item.count) {
                state.adhkarCounts[index]++;
                renderAdhkarList(); 
            }
        }

        function updateTotalProgress(current, target) {
            const bar = document.getElementById('total-adhkar-progress');
            const text = document.getElementById('total-adhkar-count');
            const percent = (current / target) * 100;
            bar.style.width = `${percent}%`;
            text.innerText = `${current}/${target}`;
            if(current >= target) bar.classList.add('bg-secondary');
        }

        async function loadHomeSections() {
            loadSection('quran');

            loadSection('lessons');

        async function loadSection(type, isLoadMore = false) {
            const config = sectionConfig[type];
            const btn = document.getElementById(config.btnId);
            const grid = document.getElementById(config.containerId);
            
            if(isLoadMore) {
                btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i>';
                btn.disabled = true;
            } else {
                grid.innerHTML = '<div class="col-span-full text-center py-10"><span class="loader"></span></div>';
                config.page = 1;
            }

            const q = encodeURIComponent(config.query);
            const url = `https://archive.org/advancedsearch.php?q=${q}&fl[]=identifier,title,downloads&sort[]=downloads+desc&rows=${config.limit}&page=${config.page}&output=json`;

            try {
                const res = await fetch(url);
                const data = await res.json();
                const docs = data.response.docs;

                if(!isLoadMore) grid.innerHTML = '';

                if(docs.length > 0) {
                    docs.forEach(doc => grid.appendChild(createAudioCard(doc, type)));
                    config.page++;
                    if(docs.length >= config.limit) btn.classList.remove('hidden');
                    else btn.classList.add('hidden');
                } else {
                    if(!isLoadMore) grid.innerHTML = '<p class="text-gray-500 col-span-full text-center text-sm">No results.</p>';
                    btn.classList.add('hidden');
                }
            } catch(e) {
                console.error(e);
                if(!isLoadMore) grid.innerHTML = '<p class="text-red-500 col-span-full text-center text-sm">Error loading data.</p>';
            } finally {
                const btnText = state.lang === 'ar' ? 'تحميل المزيد' : 'Load More';
                btn.innerHTML = `<i class="fas fa-plus-circle"></i> <span>${btnText}</span>`;
                btn.disabled = false;
            }
        }

        // --- UI Generators ---
        function createAudioCard(doc, type) {
            const id = doc.identifier;
            const imgUrl = `https://archive.org/services/img/${id}`;
            const title = doc.title.length > 40 ? doc.title.substring(0,40)+"..." : doc.title;
            
            const div = document.createElement('div');
            div.className = "bg-white dark:bg-slate-700 rounded-lg shadow hover:shadow-xl transition cursor-pointer group overflow-hidden border border-gray-100 dark:border-gray-600 animate-fade-in";
            div.onclick = () => openModal(doc);

            div.innerHTML = `
                <div class="audio-card-visual aspect-square relative">
                    <img src="${imgUrl}" class="w-full h-full object-cover absolute inset-0 z-10 opacity-90 group-hover:opacity-100 transition" onerror="this.style.display='none'; this.nextElementSibling.style.display='flex';" onload="this.nextElementSibling.style.display='none'">
                    
                    <!-- Fallback Vinyl Visual -->
                    <div class="css-audio-cover hidden z-0"></div>
                    
                    <div class="absolute inset-0 bg-black/40 flex items-center justify-center opacity-0 group-hover:opacity-100 transition z-20">
                        <div class="w-12 h-12 bg-secondary rounded-full flex items-center justify-center text-white pl-1 shadow-lg transform scale-75 group-hover:scale-100 transition">
                            <i class="fas fa-play"></i>
                        </div>
                    </div>
                </div>
                <div class="p-3 text-center">
                    <h4 class="font-bold text-primary dark:text-blue-300 text-sm h-10 overflow-hidden leading-tight font-sans">${title}</h4>
                </div>
            `;
            return div;
        }

        function openModal(doc) {
            const modal = document.getElementById('modal');
            const container = modal.querySelector('div'); 
            
            document.getElementById('modal-title').innerText = doc.title;
            document.getElementById('modal-desc').innerText = doc.description ? (Array.isArray(doc.description) ? doc.description[0] : doc.description) : "No description available.";
            document.getElementById('modal-link').href = `https://archive.org/details/${doc.identifier}`;
            
            // Set Embed Source
            const embedUrl = `https://archive.org/embed/${doc.identifier}`;
            document.getElementById('audio-embed-frame').src = embedUrl;

            modal.classList.remove('hidden');
            setTimeout(() => {
                modal.classList.remove('opacity-0');
                container.classList.remove('scale-95');
                container.classList.add('scale-100');
            }, 10);
        }

        function closeModal() {
            const modal = document.getElementById('modal');
            const container = modal.querySelector('div');
            modal.classList.add('opacity-0');
            container.classList.remove('scale-100');
            container.classList.add('scale-95');
            
            setTimeout(() => {
                document.getElementById('audio-embed-frame').src = "";
                modal.classList.add('hidden');
            }, 300);
        }
        
        document.getElementById('modal').addEventListener('click', (e) => {
            if(e.target.id === 'modal') closeModal();
        });
