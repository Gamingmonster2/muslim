    <script>
        // DOM Elements
        const cards = document.querySelectorAll('.sheikh-card');
        const iframes = [
            document.getElementById('iframe-0'),
            document.getElementById('iframe-1')
        ];
        let deferredPrompt;

        function selectSheikh(index) {
            // Update Active Card UI
            cards.forEach((card, i) => {
                if(i === index) card.classList.add('active');
                else card.classList.remove('active');
            });

            // Update Active Iframe
            iframes.forEach((iframe, i) => {
                if(i === index) iframe.classList.add('active');
                else iframe.classList.remove('active');
            });
        }

        // PWA Install Logic
        window.addEventListener('beforeinstallprompt', (e) => {
            e.preventDefault();
            deferredPrompt = e;
            document.getElementById('installBtn').style.display = 'flex';
        });

        function installApp() {
            if (deferredPrompt) {
                deferredPrompt.prompt();
                deferredPrompt.userChoice.then((choiceResult) => {
                    if (choiceResult.outcome === 'accepted') {
                        console.log('User accepted the install prompt');
                    }
                    deferredPrompt = null;
                });
            } else {
                alert('للتثبيت: اضغط على خيارات المتصفح ثم "إضافة إلى الشاشة الرئيسية"');
            }
        }

        // Initialize
        selectSheikh(0);

    </script>
